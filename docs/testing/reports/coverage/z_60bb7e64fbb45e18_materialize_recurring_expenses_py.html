1""" 

2FR6 / SASD D3: turn recurring rules into ordinary expenses on their due dates. 

3 

4Safe to run any number of times a day: each rule is locked while it is 

5processed, and an expense is created only for dates the rule has not produced 

6yet (also enforced by the unique (recurring_rule, date) constraint). Missed 

7runs are caught up on the next run. "Today" is the rule owner's local date. 

8"""

22def _on_day(year, month, anchor_day): 

23 """The anchor day in that month, clamped to the month's last day (31 Jan -> 28 Feb).""" 

24 return date(year, month, min(anchor_day, monthrange(year, month)[1]))

27def next_occurrence(value, frequency, anchor_day): 

28 if frequency == "daily": 

29 return value + timedelta(days=1) 

30 if frequency == "weekly": 

31 return value + timedelta(days=7) 

32 if frequency == "yearly": 

33 return _on_day(value.year + 1, value.month, anchor_day) 

34 month = value.month % 12 + 1 

35 year = value.year + (1 if value.month == 12 else 0) 

36 return _on_day(year, month, anchor_day)

39def due_dates(rule, today): 

40 """Every occurrence from the start date up to today and the end date, inclusive.""" 

41 last = min(today, rule.end_date) if rule.end_date else today 

42 occurrence = rule.start_date 

43 while occurrence <= last: 

44 yield occurrence 

45 occurrence = next_occurrence(occurrence, rule.frequency, rule.start_date.day)

48def materialize_rule(rule_id): 

49 """Create the missing expenses for one rule. Returns how many were created.""" 

50 with transaction.atomic(): 

51 # Lock only the rule row: the category join is nullable and PostgreSQL 

52 # refuses FOR UPDATE on the nullable side of an outer join. 

53 rule = ( 

54 RecurringRule.objects.select_for_update(of=("self",)) 

55 .select_related("category", "owner") 

56 .get(pk=rule_id) 

57 ) 

58 if not rule.is_active: 58 ↛ 59line 58 didn't jump to line 59 because the condition on line 58 was never true

59 return 0 

60 today = user_today(rule.owner) 

61 done = set(Expense.objects.filter(recurring_rule=rule).values_list("date", flat=True)) 

62 created = 0 

63 for occurrence in due_dates(rule, today): 

64 if occurrence in done: 

65 continue 

66 expense = Expense.objects.create( 

67 owner=rule.owner, 

68 recurring_rule=rule, 

69 category=rule.category, 

70 amount=rule.amount, 

71 date=occurrence, 

72 payment_method=rule.payment_method, 

73 merchant=rule.merchant, 

74 notes=rule.notes, 

75 description=rule.merchant or rule.notes, 

76 ) 

77 log_change(rule.owner, AuditLog.Action.CREATE, expense) 

78 created += 1 

79 return created
